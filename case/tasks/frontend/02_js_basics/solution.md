# Модуль 2: Основы JavaScript — Эталонное решение

## scoreboard.js

```js
"use strict";

// ============================================================
// 1. ИСХОДНЫЕ ДАННЫЕ
// ============================================================

// const — ссылка на массив не меняется (сам массив при этом мутабелен)
const participants = [
    { name: "Алина",  track: "solidity", scores: [95, 88, 92],  isActive: true  },
    { name: "Иван",   track: "frontend", scores: [70, 65, 80],  isActive: true  },
    { name: "Мария",  track: "devops",   scores: [55, 61, 58],  isActive: true  },
    { name: "Данияр", track: "solidity", scores: [],            isActive: true  }, // ещё не сдавал модули
    { name: "Олег",   track: "frontend", scores: [90, 91],      isActive: false }, // выбыл из зачёта
];

// ============================================================
// 2. getAverage — Function Declaration
// ============================================================

/**
 * Средний балл по массиву оценок.
 * @param {number[]} scores — массив баллов
 * @returns {number} среднее, округлённое до целого; 0 для пустого массива
 */
function getAverage(scores) {
    // Защита от деления на ноль: пустой массив сразу даёт 0
    if (scores.length === 0) {
        return 0;
    }

    let sum = 0; // let — значение меняется на каждой итерации

    // Классический for: индекс i живёт только внутри цикла (блочная область видимости let)
    for (let i = 0; i < scores.length; i++) {
        sum += scores[i];
    }

    return Math.round(sum / scores.length);
}

// ============================================================
// 3. getGrade — Function Expression
// ============================================================

/**
 * Буквенная оценка по среднему баллу.
 * switch(true) позволяет писать диапазоны в case вместо сравнения с константой.
 */
const getGrade = function (average) {
    switch (true) {
        case average >= 90:
            return "A";
        case average >= 75:
            return "B";
        case average >= 60:
            return "C";
        default:
            return "F";
    }
};

// ============================================================
// 4. getTrackLabel — стрелочная функция
// ============================================================

const getTrackLabel = (track) => {
    switch (track) {
        case "solidity":
            return "Смарт-контракты";
        case "frontend":
            return "Интерфейсы";
        case "devops":
            return "Инфраструктура";
        default:
            return "Не определено";
    }
};

// ============================================================
// 5. buildReport — параметр по умолчанию + возврат объекта
// ============================================================

/**
 * Собирает карточку результата участника.
 * minScore = 60 — параметр по умолчанию: подставится, если аргумент не передан.
 */
function buildReport(participant, minScore = 60) {
    const average = getAverage(participant.scores);

    return {
        name: participant.name,
        track: getTrackLabel(participant.track),
        average: average,
        grade: getGrade(average),
        // Тернарный оператор: условие ? значение_если_true : значение_если_false
        passed: average >= minScore ? true : false,
    };
}

// ============================================================
// 6. ОТЧЁТ ПО АКТИВНЫМ УЧАСТНИКАМ (for...of + continue)
// ============================================================

console.log("=== Ведомость ===");

// for...of перебирает сами элементы массива (в отличие от for...in, который даёт ключи)
for (const participant of participants) {
    if (!participant.isActive) {
        continue; // пропускаем итерацию, не прерывая весь цикл
    }

    const report = buildReport(participant);

    // Шаблонный литерал: значения подставляются через ${...} без склейки плюсами
    console.log(
        `${report.name} — ${report.track} | средний: ${report.average} | ` +
        `оценка: ${report.grade} | зачёт: ${report.passed ? "да" : "нет"}`
    );
}

// ============================================================
// 7. СТАТИСТИКА ЧЕРЕЗ while
// ============================================================

let index = 0;
let countA = 0;

// while удобен, когда счётчиком нужно управлять вручную
while (index < participants.length) {
    const p = participants[index];

    if (p.isActive && getGrade(getAverage(p.scores)) === "A") {
        countA++;
    }

    index++; // без инкремента получится бесконечный цикл
}

console.log(`\nОтличников ("A") среди активных: ${countA}`);

// ============================================================
// 8. ГРУППИРОВКА ПО НАПРАВЛЕНИЯМ (for...in)
// ============================================================

const countByTrack = {};

for (const p of participants) {
    // Если ключа ещё нет — undefined, поэтому ?? подставляет 0
    countByTrack[p.track] = (countByTrack[p.track] ?? 0) + 1;
}

console.log("\n=== Участников по направлениям ===");

// for...in перебирает ключи объекта; значение достаём через квадратные скобки
for (const track in countByTrack) {
    console.log(`${track}: ${countByTrack[track]}`);
}

// ============================================================
// 9. ТИПЫ, СРАВНЕНИЯ И ОПЕРАТОР ??
// ============================================================

console.log("\n=== Типы данных ===");
console.log(typeof "Алина");        // "string"
console.log(typeof 42);              // "number"
console.log(typeof true);            // "boolean"
console.log(typeof participants[0]); // "object" (массивы и объекты — оба "object")

console.log("\n=== == против === ===");
console.log("5" == 5);   // true  — нестрогое сравнение приводит строку к числу
console.log("5" === 5);  // false — строгое сравнение сначала сверяет тип

// -------- Было: многословная проверка на undefined --------
// let bonus;
// if (bonus === undefined || bonus === null) {
//     bonus = 0;
// }

// -------- Стало: оператор нулевого слияния --------
let bonus;
const safeBonus = bonus ?? 0; // подставит 0 только для null и undefined
console.log("\nБонус по умолчанию:", safeBonus); // 0

// Логическое присваивание: запишет значение, только если переменная null/undefined
let organizerNote;
organizerNote ??= "Замечаний нет";
console.log("Комментарий организатора:", organizerNote);

// Важное отличие от ||: ноль и пустая строка ?? считает валидными значениями
const zeroScore = 0;
console.log(zeroScore ?? 100); // 0   — ?? пропускает только null/undefined
console.log(zeroScore || 100); // 100 — || отсекает любое "ложное" значение
```

## Разбор решения

1. **Три способа объявления функций.** `getAverage` — Function Declaration (всплывает, доступна до строки объявления), `getGrade` — Function Expression (доступна только после присваивания), `getTrackLabel` — стрелочная. На практике выбор чаще стилистический, но понимать разницу в hoisting нужно.
2. **`switch(true)`.** Обычный `switch` сравнивает выражение с каждым `case` строго (`===`). Передав `true`, мы получаем возможность писать в `case` произвольные условия — компактная альтернатива длинной цепочке `if / else if`.
3. **Защита от деления на ноль.** Участник с пустым `scores` — не редкий, а ожидаемый случай. Ранний `return 0` в начале функции избавляет от `NaN` во всём остальном коде.
4. **`for` против `for...of` против `for...in`.** `for` нужен, когда важен индекс; `for...of` даёт сами элементы и читается чище; `for...in` предназначен для ключей объекта. Применять `for...in` к массиву — частая ошибка: он вернёт индексы в виде строк.
5. **`continue` вместо вложенного `if`.** Ранний пропуск итерации держит тело цикла плоским, без лишнего уровня вложенности.
6. **Параметр по умолчанию.** `minScore = 60` срабатывает, когда аргумент не передан или равен `undefined`. Это устраняет ручную проверку внутри функции.
7. **Область видимости.** `let i` внутри `for` существует только в пределах цикла; `sum` объявлена в теле функции и недоступна снаружи. Глобальных переменных в решении нет — каждая функция работает только со своими аргументами.
8. **`??` против `||`.** `||` считает «ложными» `0`, `""`, `false` и подменяет их значением по умолчанию — это классический источник багов в подсчёте баллов. `??` реагирует только на `null` и `undefined`, поэтому нулевой балл остаётся нулевым.
